24template<size_t sectorCount, size_t sectorSize = FF_MAX_SS>
25class RamIO : public IO {
26 public:
27 RamIO() {
28 memset(work_buffer, 0, sizeof(work_buffer));
29 }
30
31 ~RamIO() {
32 for (auto* ptr : sectors) {
33 free(ptr);
34 }
35 }
36
37 DSTATUS disk_initialize(BYTE pdrv) override {
38 if (pdrv != 0) return STA_NODISK;
39
40 // allocate sectors
41 if (sectors.size() == 0) {
42 for (size_t j = 0; j < sectorCount; j++) {
43 uint8_t* ptr = nullptr;
44#ifdef ESP32
45 ptr = (uint8_t*)ps_malloc(sectorSize);
46#elif defined(ARDUINO_ARCH_RP2040) && defined(RP2350_PSRAM_CS)
47 // free() releases PSRAM blocks as well
48 ptr = (uint8_t*)pmalloc(sectorSize);
49#endif
50 if (ptr == nullptr) ptr = (uint8_t*)malloc(sectorSize);
51 if (ptr == nullptr) {
52 // out of memory: release what we got and report the failure
53 for (auto* p : sectors) free(p);
54 sectors.clear();
55 status = STA_NOINIT;
56 return status;
57 }
58 memset(ptr, 0, sectorSize);
59 sectors.push_back(ptr);
60 }
61 }
62 status = STA_CLEAR;
63 return status;
64 }
65
66 DSTATUS disk_status(BYTE pdrv) override {
67 if (pdrv != 0) return STA_NODISK;
68 return status;
69 }
70
71 DRESULT disk_read(BYTE pdrv, BYTE* buffer, LBA_t sectorNo, UINT nSectors) override {
72 if (pdrv != 0) return RES_NOTRDY;
73 if (status == STA_NOINIT) return RES_NOTRDY;
74 if (sectors.size() == 0 || sectorNo + nSectors > sectorCount) return RES_ERROR;
75 for (size_t j = sectorNo; j < sectorNo + nSectors; j++) {
76 int idx = j - sectorNo;
77 uint8_t* source = sectors[j];
78 memcpy(buffer + (idx * sectorSize), source , sectorSize);
79 }
80 return RES_OK;
81 }
82
83 DRESULT disk_write(BYTE pdrv, const BYTE* buffer, LBA_t sectorNo,
84 UINT nSectors) override {
85 if (pdrv != 0) return RES_NOTRDY;
86 if (status == STA_NOINIT) return RES_NOTRDY;
87 if (sectors.size() == 0 || sectorNo + nSectors > sectorCount) return RES_ERROR;
88 for (size_t j = sectorNo; j < sectorNo + nSectors; j++) {
89 size_t idx = j - sectorNo;
90 uint8_t* target = sectors[j];
91 memcpy(target, buffer + (idx * sectorSize), sectorSize);
92 }
93 return RES_OK;
94 }
95
96 DRESULT disk_ioctl(BYTE pdrv, ioctl_cmd_t cmd, void* buffer) override {
97 DRESULT res;
98 if (pdrv) return RES_PARERR; /* Check parameter */
99 switch (cmd) {
100 case CTRL_SYNC: /* Wait for end of internal write process of the drive */
101 res = RES_OK;
102 break;
103
104 case GET_SECTOR_COUNT: { /* Get drive capacity in unit of sector (DWORD)
105 */
106 DWORD result = sectorCount;
107 memcpy(buffer, &result, (sizeof(result)));
108 res = RES_OK;
109 } break;
110
111 case GET_BLOCK_SIZE: { /* Get erase block size in unit of sector (DWORD)
112 */
113 DWORD result = 1;
114 memcpy(buffer, &result, (sizeof(result)));
115 res = RES_OK;
116 } break;
117
118 case CTRL_TRIM: { /* Erase a block of sectors (used when _USE_ERASE == 1)
119 */
120 if (buffer == nullptr || status == STA_NOINIT) {
121 res = RES_PARERR;
122 break;
123 }
124 DWORD range[2];
125 // determine range
126 memcpy(&range, buffer, sizeof(range));
127 // validate range: must be inside the drive and allocated
128 if (range[0] > range[1] || range[1] >= sectorCount ||
129 sectors.size() != sectorCount) {
130 res = RES_PARERR;
131 break;
132 }
133 // clear memory
134 for (DWORD j = range[0]; j <= range[1]; j++) {
135 memset(sectors[j], 0, sectorSize);
136 }
137 res = RES_OK; /* FatFs does not check result of this command */
138 } break;
139
140 default:
141 res = RES_PARERR;
142 break;
143 }
144 return res;
145 }

147 virtual FRESULT mount(FatFs& fs, BYTE ldrv) override
148 {
149 memset(work_buffer, 0, sizeof(work_buffer));
150
151 // the file system is empty so we need to format it
152 char path[6] = {0};
153 snprintf(path, sizeof(path), "%d:", ldrv);
154 FRESULT rc = fs.f_mkfs(path, nullptr, work_buffer, sizeof(work_buffer));
155 if (FR_OK != rc) {
156 return rc;
157 }
158
159 return fs.f_mount(&fatfs, path, 0);
160 }
161
162 protected:
163#ifdef ETL_NO_STL
164 vector<uint8_t*, sectorCount> sectors;
165#else
166 vector<uint8_t*> sectors;
167#endif
168 DSTATUS status = STA_NOINIT;
169 uint8_t work_buffer[FF_MAX_SS];
170};
