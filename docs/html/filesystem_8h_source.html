41class directory_iterator {
42 public:
43 directory_iterator() : end_flag(true) {}
44 explicit directory_iterator(const string& rootPath)
45 : directory_iterator(get_fatfs(), rootPath) {}
47 directory_iterator(fatfs::FatFs* fsPtr, const string& rootPath)
48 : root(rootPath), end_flag(false) {
49 if (!fsPtr) {
50 end_flag = true;
51 return;
52 }
53 this->fs = fsPtr;
54
55 // Open directory using FatFs API
56 FRESULT res = fs->f_opendir(&dir, root.empty() ? "/" : root.c_str());
57 if (res != FR_OK) {
58 end_flag = true;
59 } else {
60 advance();
61 }
62 }

65 if (!end_flag && fs) {
66 fs->f_closedir(&dir);
67 }
68 }

71 advance();
72 return *this;
73 }

93 if (!fs) {
94 end_flag = true;
95 return;

136class recursive_directory_iterator {
137 public:
138 recursive_directory_iterator() : end_flag(true) {}
139 explicit recursive_directory_iterator(const string& rootPath)
140 : recursive_directory_iterator(get_fatfs(), rootPath) {}
142 recursive_directory_iterator(fatfs::FatFs* fsPtr, const string& rootPath)
143 : end_flag(false) {
144 if (!fsPtr) {
145 end_flag = true;
146 return;
147 }
148 this->fs = fsPtr;
149
150 // Start with root directory
151 if (!push_directory(rootPath.empty() ? "/" : rootPath)) {
152 end_flag = true;
153 } else {
154 advance();
155 }
156 }
157
158 ~recursive_directory_iterator() {
159 // Close all open directories
160 while (!stack.empty()) {
161 if (fs) fs->f_closedir(&stack.back().dir);
162 stack.pop_back();
163 }
164 }
165
166 recursive_directory_iterator& operator++() {
167 advance();
168 return *this;
169 }
170
171 directory_entry operator*() const { return current_entry; }
172
173 bool operator!=(const recursive_directory_iterator& other) const {
174 return end_flag != other.end_flag;
175 }
176
177 static recursive_directory_iterator end() {
178 return recursive_directory_iterator();
179 }
180
181 private:
182 struct DirLevel {
183 DIR dir;
184 string path;
185 };
186
187 bool push_directory(const string& path) {
188 if (!fs) return false;
189
190 DirLevel level;
191 level.path = path;
192 FRESULT res = fs->f_opendir(&level.dir, path.c_str());
193 if (res == FR_OK) {
194 stack.push_back(level);
195 return true;
196 }
197 return false;
198 }
199
200 void advance() {
201 if (!fs || stack.empty()) {
202 end_flag = true;
203 return;
204 }
205
206 FILINFO info;
207
208 while (!stack.empty()) {
209 auto& level = stack.back();
210
211 // Read next entry in current directory
212 FRESULT res = fs->f_readdir(&level.dir, &info);
213
214 if (res != FR_OK || info.fname[0] == 0) {
215 // No more entries in this directory, pop and continue with parent
216 fs->f_closedir(&level.dir);
217 stack.pop_back();
218 continue;
219 }
220
221 // Skip . and .. entries
222 if (strcmp(info.fname, ".") == 0 || strcmp(info.fname, "..") == 0) {
223 continue;
224 }
225
226 // Build full path for current entry
227 current_entry.path = level.path;
228 if (!current_entry.path.empty() && current_entry.path.back() != '/') {
229 current_entry.path += "/";
230 }
231 current_entry.path += info.fname;
232 current_entry.is_directory = (info.fattrib & AM_DIR) != 0;
233 current_entry.size = info.fsize;
234
235 // If this is a directory, push it onto the stack for depth-first
236 // traversal
237 if (current_entry.is_directory) {
238 push_directory(current_entry.path);
239 }
240
241 // Return current entry (directory or file)
242 return;
243 }
244
245 // Stack is empty, we're done
246 end_flag = true;
247 }
248
249 fatfs::FatFs* fs = nullptr;
250 // FAT32 has no strict limit for directory depth level, let's limit this implementation
251#ifdef ETL_NO_STL
252 vector<DirLevel, 255> stack;
253#else
254 vector<DirLevel> stack;
255#endif
256 directory_entry current_entry;
257 bool end_flag{true};
258};
